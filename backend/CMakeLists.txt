cmake_minimum_required(VERSION 3.22)
project(osrs_signature_backend VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
find_package(Threads REQUIRED)
find_package(CURL REQUIRED)
find_path(ASIO_INCLUDE_DIR asio.hpp REQUIRED)
FetchContent_Declare(crow_headers
    URL https://codeload.github.com/CrowCpp/Crow/tar.gz/refs/tags/v1.2.1)
FetchContent_GetProperties(crow_headers)
if(NOT crow_headers_POPULATED)
    FetchContent_Populate(crow_headers)
endif()
FetchContent_Declare(json
    URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz)
FetchContent_MakeAvailable(json)
add_executable(osrs-signature-backend src/main.cpp)
target_compile_definitions(osrs-signature-backend PRIVATE ASIO_STANDALONE)
target_include_directories(osrs-signature-backend PRIVATE
    ${crow_headers_SOURCE_DIR}/include ${ASIO_INCLUDE_DIR})
target_link_libraries(osrs-signature-backend PRIVATE
    CURL::libcurl Threads::Threads nlohmann_json::nlohmann_json)
enable_testing()
add_executable(domain-tests tests/domain_test.cpp)
target_include_directories(domain-tests PRIVATE src)
add_test(NAME domain-tests COMMAND domain-tests)

add_executable(memory-service-tests tests/memory_service_test.cpp)
target_compile_definitions(memory-service-tests PRIVATE ASIO_STANDALONE)
target_include_directories(memory-service-tests PRIVATE ${crow_headers_SOURCE_DIR}/include ${ASIO_INCLUDE_DIR})
target_link_libraries(memory-service-tests PRIVATE CURL::libcurl Threads::Threads nlohmann_json::nlohmann_json)
target_compile_options(memory-service-tests PRIVATE -UNDEBUG)
target_compile_options(domain-tests PRIVATE -UNDEBUG)
add_test(NAME memory-service-tests COMMAND memory-service-tests)
set_tests_properties(memory-service-tests PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}")
